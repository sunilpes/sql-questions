Problem Statement: 
Write a SQL query to find the maximum and
minimum values of continuous ‘Sequence’ in each ‘Group’

Answers:
WITH group_split as (
	select seq, grp,
	ROW_NUMBER() OVER(partition by grp order by seq) as rnk,
	seq - ROW_NUMBER() OVER(partition by grp order by seq) as grp_split
	from main.Emp
),
final_res as (select 
	grp,
	grp_split,
	MAX(seq) as max_seq,
	MIN(seq) as min_seq
from group_split
group by 1,2)
select grp, max_seq, min_seq
from final_res;
